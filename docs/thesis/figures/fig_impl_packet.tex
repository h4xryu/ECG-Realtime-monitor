% fig_impl_packet.tex
% HiCardi BLE frame (42 bytes) and 7+7-bit ECG sample decoding.
% The decoding equation and the MSB convention are stated in the body text.
\begin{tikzpicture}[>=Stealth, thick, x=1mm, y=1mm,
  cell/.style={draw, thick, minimum height=10mm, align=center, inner sep=1pt, anchor=west},
  sync/.style={cell, fill=gray!25},
  bit/.style={draw, thick, minimum width=6mm, minimum height=6mm, inner sep=0pt, anchor=west},
  zero/.style={bit, fill=gray!25},
  off/.style={anchor=south, inner sep=1pt}]

  %% ---------------- frame ----------------
  \node[sync, minimum width=12mm] (c0) at (0,0)        {Sync\\\texttt{0x80}};
  \node[cell, minimum width=10mm] (c1) at (c0.east)    {Seq\\1\,B};
  \node[cell, minimum width=14mm] (c2) at (c1.east)    {Header\\3\,B};
  \node[cell, minimum width=16mm] (c3) at (c2.east)    {Resp.\\3\,B};
  \node[cell, minimum width=46mm, very thick] (c4) at (c3.east)
                                                      {ECG: $10\times$[tag, hi, lo]\\30\,B};
  \node[cell, minimum width=18mm] (c5) at (c4.east)    {Telemetry\\3\,B};
  \node[sync, minimum width=12mm] (c6) at (c5.east)    {Sync\\\texttt{0x8F}};

  %% byte offsets
  \foreach \c/\o in {c0/0, c1/1, c2/2, c3/5, c4/8, c5/38, c6/41}
    \node[off] at (\c.north west) {\o};
  \node[off] at (c6.north east) {42};

  %% ---------------- one ECG triplet ----------------
  \node[bit, minimum width=24mm, very thick] (tag) at (4,-26) {tag \texttt{0xF0}/\texttt{0xFF}};

  \node[zero] (h0) at ([xshift=4mm]tag.east) {0};
  \foreach \i/\b in {1/13,2/12,3/11,4/10,5/9,6/8,7/7} {
    \pgfmathtruncatemacro{\p}{\i-1}
    \node[bit] (h\i) at (h\p.east) {$b_{\b}$};
  }
  \node[zero] (l0) at ([xshift=4mm]h7.east) {0};
  \foreach \i/\b in {1/6,2/5,3/4,4/3,5/2,6/1,7/0} {
    \pgfmathtruncatemacro{\p}{\i-1}
    \node[bit] (l\i) at (l\p.east) {$b_{\b}$};
  }
  \node[off] at (h4.north) {hi byte};
  \node[off] at (l4.north) {lo byte};

  %% ---------------- link the first triplet to the expansion ----------------
  \draw[densely dashed] (c4.south west) -- (tag.north west);
  \draw[densely dashed] ([xshift=4.6mm]c4.south west) -- (l7.north east);
\end{tikzpicture}
